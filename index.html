async function loadQuizzes() { try { const res = await fetch('./quizzes.json'); if (!res.ok) throw new Error(`HTTP ${res.status}`); const data = await res.json(); // Support both wrapped metadata JSON and direct array JSON quizzes = Array.isArray(data) ? data : (data.quizzes || []); if (quizzes.length > 0) { renderQuestion(); } else { document.getElementById('question').innerText = 'No questions available in quizzes.json.'; } } catch (err) { console.error('Error fetching quiz data:', err); document.getElementById('question').innerText = 'Failed to load quizzes.json. Make sure the file exists on GitHub Pages.'; } }
